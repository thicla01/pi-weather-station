# Mise à jour Pi 192.168.6.49 — 2026-04-15
# Corrections : timeouts axios + capture des logs serveur + timestamps dans les logs

# 1. Récupérer les dernières modifications (inclut les timeouts axios)
cd ~/pi-weather-station
git pull

# 2. Ajouter la capture des logs dans un drop-in systemd
#    (crée le dossier s'il n'existe pas, puis écrit le fichier)
mkdir -p ~/.config/systemd/user/pi-weather-server.service.d
cat > ~/.config/systemd/user/pi-weather-server.service.d/logging.conf << 'EOF'
[Service]
StandardOutput=append:/home/pi/pi-weather-station/server.log
StandardError=append:/home/pi/pi-weather-station/server.log
EOF

# 3. Recharger systemd et redémarrer le serveur
systemctl --user daemon-reload
systemctl --user restart pi-weather-server

# 4. Vérifier que le serveur a bien démarré
systemctl --user status pi-weather-server --no-pager

# 5. Vérifier que les logs sont capturés
tail -5 ~/pi-weather-station/server.log
